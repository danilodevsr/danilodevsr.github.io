⚙️ Configuración del Motor

index.js AETHER 2.0 SYNTAX 
import { AetherScene, Laptop, DeviceOrbit } from 'aether-3d';

// 1. Inicializar escena reactiva al scroll
const scene = new AetherScene({ canvas: '#webgl-canvas' });

// 2. Añadir Laptop que se abre al hacer scroll (0% -> 50%)
const laptop = scene.add(new Laptop({ 
  openOnScroll: [0, 0.5],
  lightColor: '#00f0ff' 
}));

// 3. Emitir luz y revelar el Celular, Servidor y Tierra (50% -> 100%)
laptop.onOpenEmit(new DeviceOrbit([
  'smartphone', 'server-tower', 'cyber-earth'
]));

scene.render();
